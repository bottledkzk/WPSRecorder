cmake_minimum_required(VERSION 3.16)
project(wpsrecorder VERSION 2.3.13 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "wpsrecorder requires Windows 10 or newer")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Network Multimedia MultimediaWidgets Concurrent)
if(TARGET Qt5::qmake)
    get_target_property(recorder_qmake Qt5::qmake IMPORTED_LOCATION)
    if(recorder_qmake AND EXISTS "${recorder_qmake}")
        execute_process(COMMAND "${recorder_qmake}" -query QT_INSTALL_TRANSLATIONS
            OUTPUT_VARIABLE recorder_qt_translations OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    endif()
endif()
find_file(RECORDER_QT_ZH_QM NAMES qtbase_zh_CN.qm
    HINTS "${recorder_qt_translations}" "${Qt5Core_DIR}/../../qt5/translations" "${Qt5Core_DIR}/../../../translations")
if(NOT RECORDER_QT_ZH_QM)
    message(FATAL_ERROR "Install Qt 5 translations (vcpkg qt5-translations) or set RECORDER_QT_ZH_QM.")
endif()
configure_file(i18n/ui_translations.qrc.in ui_translations.qrc @ONLY)
add_library(recorder_core STATIC
    src/recordingconfig.cpp src/recordingclock.cpp src/framecapture.cpp src/desktopcapture.cpp src/desktopcolor.cpp src/windowcapture.cpp
    src/audiocapture.cpp src/cameracapture.cpp src/compositor.cpp src/previewchannel.cpp src/recordingengine.cpp src/controlserver.cpp src/mediaeditor.cpp src/editproject.cpp src/speech.cpp src/editassets.qrc
    src/recordingcheck.cpp src/recordingmarkers.cpp src/recordingrecovery.cpp src/speechservices.cpp src/dubbing.cpp src/smartcut.cpp
    src/recordingassistanttools.cpp src/recordingassistant.cpp src/recording_assistant.qrc
    src/subtitletext.cpp src/subtitlereview.cpp src/recordingpixels.cpp src/recordingframestream.cpp src/chapternarration.cpp src/editsequence.cpp
    src/sequencesmartcut.cpp src/windowidentity.cpp
    src/uilanguage.cpp "${CMAKE_CURRENT_BINARY_DIR}/ui_translations.qrc")
target_include_directories(recorder_core PUBLIC src)
target_compile_definitions(recorder_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(recorder_core PUBLIC /utf-8 /W4 /permissive- /EHsc /Zc:__cplusplus)
target_link_libraries(recorder_core PUBLIC Qt5::Core Qt5::Gui Qt5::Network
    user32 gdi32 ole32 uuid avrt psapi d3d11 d3dcompiler dxgi mfplat mfreadwrite mfuuid mf advapi32 shell32 propsys dwmapi windowsapp)
add_executable(wpsrecorder_cli src/climain.cpp src/recordertasks.cpp)
target_link_libraries(wpsrecorder_cli PRIVATE recorder_core)
add_library(recorder_edit_ui STATIC src/clipeditor.cpp src/editanalysis.cpp src/editwidgets.cpp src/editplayer.cpp src/editsequencedialog.cpp
    src/submissiondialog.cpp src/subtitledialog.cpp src/subtitlereviewdialog.cpp
    src/speechsettingsdialog.cpp src/smartcutdialog.cpp src/sequencesmartcutdialog.cpp src/playerwindow.cpp src/previewvideo.cpp)
target_link_libraries(recorder_edit_ui PUBLIC recorder_core Qt5::MultimediaWidgets)
set(RECORDER_WORKBENCH_SOURCES src/mainwindow.cpp src/studioworkspace.cpp src/regionselector.cpp src/minimalrecorder.cpp
    src/annotationoverlay.cpp src/recordingwidgets.cpp src/recordingcheckdialog.cpp src/recordingassistantdialog.cpp src/assistanthistory.cpp src/studioassistant.cpp src/uitheme.cpp src/theme.qrc)
add_executable(wpsrecorder WIN32 src/main.cpp src/studioapp.cpp src/appicon.rc ${RECORDER_WORKBENCH_SOURCES})
target_link_libraries(wpsrecorder PRIVATE recorder_edit_ui Qt5::MultimediaWidgets Qt5::Concurrent shell32)
set(RECORDER_WEBVIEW2_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../../thirdparty_install/webview2-1.0.2903.40"
    CACHE PATH "Path to the extracted Microsoft.Web.WebView2 SDK")
if(NOT EXISTS "${RECORDER_WEBVIEW2_SDK}/build/native/include/WebView2.h")
    message(FATAL_ERROR "Set RECORDER_WEBVIEW2_SDK to the Microsoft.Web.WebView2 SDK directory.")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(recorder_webview_arch x64)
else()
    set(recorder_webview_arch x86)
endif()
target_sources(wpsrecorder PRIVATE src/webstudiosurface.cpp src/webstudio.cpp src/webstudioedit.cpp)
target_include_directories(wpsrecorder SYSTEM PRIVATE "${RECORDER_WEBVIEW2_SDK}/build/native/include")
target_link_libraries(wpsrecorder PRIVATE
    "${RECORDER_WEBVIEW2_SDK}/build/native/${recorder_webview_arch}/WebView2LoaderStatic.lib"
    shlwapi version)
target_compile_definitions(wpsrecorder PRIVATE WPS_WEB_STUDIO)
add_custom_target(recorder_studio_assets
    COMMAND "${CMAKE_COMMAND}" -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/studio" "${CMAKE_BINARY_DIR}/bin/$<$<BOOL:${CMAKE_CONFIGURATION_TYPES}>:$<CONFIG>/>studio")
add_dependencies(wpsrecorder recorder_studio_assets)
install(DIRECTORY studio/ DESTINATION bin/studio)
foreach(target wpsrecorder wpsrecorder_cli)
    target_compile_definitions(${target} PRIVATE WPSRECORDER_VERSION="${PROJECT_VERSION}")
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
install(TARGETS wpsrecorder wpsrecorder_cli RUNTIME DESTINATION bin)
install(FILES scripts/audio8_worker.py DESTINATION bin/tts)
install(DIRECTORY src/assets/audio8-voices/ DESTINATION bin/tts/voices)
add_custom_command(TARGET wpsrecorder_cli POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:wpsrecorder_cli>/tts"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/scripts/audio8_worker.py" "$<TARGET_FILE_DIR:wpsrecorder_cli>/tts/audio8_worker.py"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/src/assets/audio8-voices" "$<TARGET_FILE_DIR:wpsrecorder_cli>/tts/voices")
